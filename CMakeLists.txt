cmake_minimum_required(VERSION 3.22)
project(stm32h745-disco C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(HOST_TESTS "Build host unit tests" OFF)
option(HOST_SIM "Build host LVGL SDL simulator" OFF)
option(COVERAGE "Enable gcov on host tests" OFF)
set(CORE "" CACHE STRING "Firmware core: M7, M4, BOTH, or empty for BOTH")

if(HOST_TESTS AND HOST_SIM)
    message(FATAL_ERROR "HOST_TESTS and HOST_SIM are mutually exclusive")
endif()

if(HOST_TESTS)
    include(${CMAKE_SOURCE_DIR}/cmake/HostTests.cmake)
elseif(HOST_SIM)
    include(${CMAKE_SOURCE_DIR}/cmake/HostSim.cmake)
else()
    include(${CMAKE_SOURCE_DIR}/cmake/Firmware.cmake)
    if(CORE STREQUAL "" OR CORE STREQUAL "BOTH")
        stm32_add_firmware(M7)
        stm32_add_firmware(M4)
        add_custom_target(firmware ALL DEPENDS firmware-m7 firmware-m4)
    elseif(CORE STREQUAL "M7")
        stm32_add_firmware(M7)
    elseif(CORE STREQUAL "M4")
        stm32_add_firmware(M4)
    else()
        message(FATAL_ERROR "Configure with -DHOST_TESTS=ON, -DHOST_SIM=ON, or -DCORE=M7|M4|BOTH")
    endif()
endif()
